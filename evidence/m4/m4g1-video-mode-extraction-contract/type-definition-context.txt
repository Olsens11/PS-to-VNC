   161   * D17AL-F1:
   162   *
   163   * GS FRAME/DISPFB widths are expressed in 64-pixel units.  The historical
   164   * gsHires code truncated Width / 64, which makes non-64-multiple modes such
   165   * as 720 and 800 pixels wrap across scanlines.
   166   *
   167   * The EE/RFB image remains tightly packed at the visible width.  Only GS
   168   * presentation storage uses this rounded pitch.
   169   */
   170  static u32 ps2vnc_hires_pitch_pixels(const GSGLOBAL *gsGlobal)
   171  {
   172  	return ((u32)gsGlobal->Width + 63u) & ~63u;
   173  }

   180  static void gsKit_setpass(GSGLOBAL *gsGlobal, const SPass * __pass, int yoff)
   181  {
   182  	u64 *p_data;
   183  	u64 *p_store;
   184  	int qsize = 8;
   185  
   186  	p_data = p_store = (u64*)gsKit_heap_alloc(gsGlobal, qsize, (qsize * 16), GIF_AD);
   187  
   188  	*p_data++ = GIF_TAG_AD(qsize);
   189  	*p_data++ = GIF_AD;
   190  
   191  	// Context 1
   192  	*p_data++ = GS_SETREG_SCISSOR_1(0, gsGlobal->Width - 1, 0, __pass->DrawBuffer->bufferline_end - __pass->DrawBuffer->bufferline_begin);
   193  	*p_data++ = GS_SCISSOR_1;
   194  	*p_data++ = GS_SETREG_FRAME_1(__pass->DrawBuffer->addr / 8192, ps2vnc_hires_pitch_pixels(gsGlobal) / 64, gsGlobal->PSM, 0);
   195  	*p_data++ = GS_FRAME_1;
   196  	*p_data++ = GS_SETREG_ZBUF_1(__pass->DepthBuffer->addr / 8192, gsGlobal->PSMZ, (gsGlobal->ZBuffering == GS_SETTING_ON) ? 0 : 1);
   197  	*p_data++ = GS_ZBUF_1;
   198  	*p_data++ = GS_SETREG_XYOFFSET_1(gsGlobal->OffsetX, yoff + gsGlobal->OffsetY + __pass->DrawBuffer->bufferline_begin*16);
   199  	*p_data++ = GS_XYOFFSET_1;
   200  
   201  	// Context 2
   202  	*p_data++ = GS_SETREG_SCISSOR_1(0, gsGlobal->Width - 1, 0, __pass->DrawBuffer->bufferline_end - __pass->DrawBuffer->bufferline_begin);
   203  	*p_data++ = GS_SCISSOR_2;
   204  	*p_data++ = GS_SETREG_FRAME_1(__pass->DrawBuffer->addr / 8192, ps2vnc_hires_pitch_pixels(gsGlobal) / 64, gsGlobal->PSM, 0);
   205  	*p_data++ = GS_FRAME_2;
   206  	*p_data++ = GS_SETREG_ZBUF_1(__pass->DepthBuffer->addr / 8192, gsGlobal->PSMZ, (gsGlobal->ZBuffering == GS_SETTING_ON) ? 0 : 1);
   207  	*p_data++ = GS_ZBUF_2;
   208  	*p_data++ = GS_SETREG_XYOFFSET_1(gsGlobal->OffsetX, yoff + gsGlobal->OffsetY + __pass->DrawBuffer->bufferline_begin*16);
   209  	*p_data++ = GS_XYOFFSET_2;
   210  }

   231  static void gsKit_create_pass(GSGLOBAL *gsGlobal, const SPass * __pass, GSQUEUE * Queue, int yoff, u32 source_yoff)
   232  {
   233  	GSQUEUE * Queue_backup = gsGlobal->CurQueue;
   234  
   235  	// Create new queue (if not already created)
   236  	if (Queue->pool[0] == NULL)
   237  		gsKit_queue_init(gsGlobal, Queue, GS_PERSISTENT, 8*1024);
   238  	gsKit_queue_set(gsGlobal, Queue);
   239  	gsKit_queue_reset(Queue);
   240  
   241  	// Set the rendering context
   242  	gsKit_setpass(gsGlobal, __pass, yoff);
   243  
   244  	// Clear depth buffer (and background)
   245  	if ((gsGlobal->ZBuffering == GS_SETTING_ON) || (__pTexture == NULL))
   246  		gsKit_clear(gsGlobal, GS_SETREG_RGBAQ(0x00,0x00,0x00,0x00,0x00));
   247  
   248  	// Fill with background image
   249  	if (__pTexture != NULL)
   250  		gsKit_setpass_background(gsGlobal, __pass, __pTexture, source_yoff);
   251  
   252  	// Restore previous queue
   253  	gsKit_queue_set(gsGlobal, Queue_backup);
   254  }

   256  void _gsKit_create_passes(GSGLOBAL *gsGlobal)
   257  {
   258  	u32 iPass;
   259  
   260  	// Create draw queues to setup the rendering for each pass
   261  	if ((gsGlobal->Interlace == GS_INTERLACED) && (gsGlobal->Field == GS_FRAME)) {
   262  		for (iPass = 0; iPass < __iPassCount; iPass++) {
   263  			// ODD lines
   264  			gsKit_create_pass(gsGlobal, &__pass[iPass], &__pass[iPass].Queue, 0, 0);
   265  			// EVEN lines, half a pixel DOWN; source is second field half
   266  			gsKit_create_pass(gsGlobal, &__pass[iPass], &__pass[iPass].QueueIF, 8, gsGlobal->Height);
   267  		}
   268  	}
   269  	else {
   270  		for (iPass = 0; iPass < __iPassCount; iPass++) {
   271  			gsKit_create_pass(gsGlobal, &__pass[iPass], &__pass[iPass].Queue, 0, 0);
   272  		}
   273  	}
   274  }

   306  // State tracking for rate limiting

   308  void gsKit_hires_flip_ext(GSGLOBAL *gsGlobal, GSFLIP_MODE mode)
   309  {
   310  	u32 iY;
   311  	u32 vsync_end;
   312  
   313  	// Handle vsync waiting based on mode
   314  	switch (mode) {
   315  		case GSFLIP_DIRECT:       vsync_end = __vsync_count; break;
   316  		case GSFLIP_VSYNC:        vsync_end = __vsync_count + 1; break;
   317  		case GSFLIP_RATE_LIMIT_1: vsync_end = __last_flip_vsync_count + 1; break;
   318  		case GSFLIP_RATE_LIMIT_2: vsync_end = __last_flip_vsync_count + 2; break;
   319  	}
   320  
   321  	// Wait for required vsyncs
   322  	while (__vsync_count < vsync_end) {
   323  		WaitSema(__sema_vsync_id);
   324  	}
   325  
   326  	// Update last flip vsync count
   327  	__last_flip_vsync_count = __vsync_count;
   328  
   329  	// HACK: The start of the first displayed line
   330  	// this is needed by the hsync interrupt but must be
   331  	// updated every time StartYOffset changes
   332  	iY = gsGlobal->StartY + gsGlobal->StartYOffset + 4;
   333  	if (gsGlobal->Interlace == GS_INTERLACED)
   334  		iY /= 2;
   335  	__iYStart = iY;
   336  
   337  	// Toggle draw queue
   338  	WaitSema(__sema_queue_id);
   339  	dmaKit_wait_fast();
   340  	__iCurrentDrawQueue = (__iCurrentDrawQueue == 0) ? 1 : 0;
   341  	SignalSema(__sema_queue_id);
   342  
   343  	gsKit_queue_reset(&__DrawQueue[__iCurrentDrawQueue]);
   344  	gsKit_queue_set(gsGlobal, &__DrawQueue[__iCurrentDrawQueue]);
   345  }

   348  #if F_gsKit_hires_prepare_bg
   349  static void gsKit_texture_to_interlaced_frame(GSTEXTURE * tex)
   350  {
   351  	int y;
   352  	int bpp;
   353  	int size;
   354  	u16 *tex_new;
   355  	u16 *tex_old;
   356  
   357  	switch (tex->PSM)
   358  	{
   359  		case GS_PSM_CT16S: bpp = 2; break;
   360  		case GS_PSM_CT24:  bpp = 3; break;
   361  		default: return; // not supported
   362  	};
   363  
   364  	size = tex->Width * tex->Height * bpp;
   365  	tex_new = (u16*)memalign(128, size);
   366  	tex_old = (u16*)tex->Mem;
   367  
   368  	for(y=0; y < tex->Height; y+=2) {
   369  		// Copy 0,2,4... to 0,1,2...
   370  		memcpy(&tex_new[(y/2)*tex->Width], &tex_old[y*tex->Width], tex->Width * bpp);
   371  		// Copy 1,3,5... to (height/2) + 0,1,2...
   372  		memcpy(&tex_new[((y+tex->Height)/2)*tex->Width], &tex_old[(y+1)*tex->Width], tex->Width * bpp);
   373  	}
   374  
   375  	free(tex->Mem);
   376  	tex->Mem = (void*)tex_new;
   377  }

   430  static int hsync_callback(int cause)
   431  {
   432  	u32 iPass;
   433  
   434  	/*
   435  	 * Keep HSync delivery suppressed while this INTC_GS callback is
   436  	 * active.  See docs/reference/EE_GS_INTERRUPT_HANDLER_GUIDE.md.
   437  	 */
   438  	*GS_IMR = 0x00007f00ULL;
   439  
   440  	if (*GS_CSR & (1<<2)) {
   441  		*GS_CSR = (1<<2);
   442  		__hsync_count++;
   443  	}
   444  
   445  	if (*GS_CSR & (1<<3)) {
   446  		*GS_CSR = (1<<3);
   447  		__vsync_count++;
   448  		__hsync_count=0;
   449  	}
   450  
   451  	for (iPass = 0; iPass < __iPassCount; iPass++) {
   452  		if (__hsync_count == (__iYStart + __pass[iPass].CRTCBuffer->scanline_begin)) {
   453  			if (iPass == 0) {
   454  				// We start rendering the first pass when the CRTC starts
   455  				// displaying the last pass of the previous field.
   456  				// So invert to get the current render field.
   457  				__iCurrentRenderField = (((GSREG*)GS_CSR)->FIELD == 1) ? 0 : 1;
   458  			}
   459  
   460  			__iCurrentPass = iPass;
   461  			iSignalSema(__sema_hsync_id);
   462  
   463  #ifdef BG_DEBUG
   464  			GS_SET_BGCOLOR((iPass+1)*63, (iPass+1)*63, (iPass+1)*63);
   465  #endif
   466  
   467  			break;
   468  		}
   469  	}
   470  
   471  
   472  	/* Restore HSync delivery immediately before interrupt retirement. */
   473  	*GS_IMR = 0x00007b00ULL;
   474  
   475  	ExitHandler();
   476  	return 0;
   477  }

   479  static void gsKit_queue_send(GSGLOBAL *gsGlobal, GSQUEUE *Queue)
   480  {
   481  	if(Queue->tag_size == 0)
   482  		return;
   483  
   484  	*(u64 *)Queue->dma_tag = DMA_TAG(Queue->tag_size, 0, DMA_END, 0, 0, 0);
   485  
   486  	if(Queue->last_type != GIF_AD)
   487  	{
   488  		*(u64 *)Queue->last_tag = ((u64)Queue->same_obj | *(u64 *)Queue->last_tag);
   489  	}
   490  
   491  	dmaKit_wait_fast();
   492  	dmaKit_send_chain_ucab(DMA_CHANNEL_GIF, Queue->pool[Queue->dbuf]);
   493  }

   495  static void dma_to_gs_thread(void * data)
   496  {
   497  	GSGLOBAL *gsGlobal = (GSGLOBAL *)data;
   498  	u32 iQueue;
   499  
   500  	while(1)
   501  	{
   502  		// Make sure the sema count == 0
   503  		while (PollSema(__sema_hsync_id) >= 0)
   504  			;
   505  
   506  		// Wait for CRTC
   507  		WaitSema(__sema_hsync_id);
   508  
   509  		WaitSema(__sema_queue_id);
   510  
   511  #ifdef BG_DEBUG
   512  		GS_SET_BGCOLOR(128,0,0);
   513  #endif
   514  
   515  		// Send current pass queue
   516  		if ((gsGlobal->Interlace == GS_INTERLACED) && (gsGlobal->Field == GS_FRAME)) {
   517  			if (__iCurrentRenderField == 1)
   518  				gsKit_queue_send(gsGlobal, &__pass[__iCurrentPass].Queue); // Odd
   519  			else
   520  				gsKit_queue_send(gsGlobal, &__pass[__iCurrentPass].QueueIF); // Even
   521  		}
   522  		else {
   523  			gsKit_queue_send(gsGlobal, &__pass[__iCurrentPass].Queue);
   524  		}
   525  
   526  		// Send current frame queue
   527  		iQueue = (__iCurrentDrawQueue == 0) ? 1 : 0;
   528  		gsKit_queue_send(gsGlobal, &__DrawQueue[iQueue]);
   529  
   530  #ifdef BG_DEBUG
   531  		GS_SET_BGCOLOR(0,128,0);
   532  #endif
   533  
   534  		SignalSema(__sema_queue_id);
   535  
   536  		// If this was the last pass, send vsync
   537  		if (__iCurrentPass == (__iPassCount-1))
   538  			SignalSema(__sema_vsync_id);
   539  	}
   540  }

   542  void gsKit_hires_init_screen(GSGLOBAL *gsGlobal, int passCount)
   543  {
   544  	/*
   545  	 * TEST15E1-A:
   546  	 * gsHires was originally initialized once per process. Runtime video
   547  	 * modes require a complete second initialization after deinit, so do
   548  	 * not inherit queue, field, texture, counter or kernel-object state
   549  	 * from the previous display instance.
   550  	 */
   551  	__iCurrentDrawQueue = 0;
   552  	__pTexture = NULL;
   553  	__iYStart = 0;
   554  	__hsync_callback_id = -1;
   555  	__vsync_count = 0;
   556  	__hsync_count = 0;
   557  	__last_flip_vsync_count = 0;
   558  	__iCurrentPass = 0;
   559  	__iCurrentRenderField = 0;
   560  	__sema_vsync_id = -1;
   561  	__sema_hsync_id = -1;
   562  	__sema_queue_id = -1;
   563  	__ThreadId = -1;
   564  	u8 ZBuffering_backup;
   565  	ee_sema_t sema;
   566  	u32 iPass;
   567  	u32 iBufferline;
   568  	u32 iPassHeight;
   569  	u32 iBpp;
   570  	u32 iHeightAlign;
   571  	u32 iPitchPixels;
   572  	ee_thread_t thread;
   573  
   574  	// Limit __iPassCount between 2 and 4:
   575  	// 2 Passes: Front buffer split in 2, used as front and back buffer. Separate depth buffer (half the size) needed.
   576  	// 3 Passes: Front buffer split in 3, used as front, back and depth buffer.
   577  	// 4 Passes: Front buffer split in 4, used as front, back and depth buffer + some temporary texture space (currently not used).
   578  	__iPassCount = passCount;
   579  	__iPassCount = __iPassCount < 2 ? 2 : __iPassCount;
   580  	__iPassCount = __iPassCount > 4 ? 4 : __iPassCount;
   581  
   582  	if (__iPassCount > 2) {
   583  		// Depth buffer uses the same buffers as the color buffer, so fix PSMZ
   584  		switch (gsGlobal->PSM) {
   585  			case GS_PSM_CT32:  gsGlobal->PSMZ = GS_PSMZ_32;  break;
   586  			case GS_PSM_CT24:  gsGlobal->PSMZ = GS_PSMZ_32;  break;
   587  			case GS_PSM_CT16:  gsGlobal->PSMZ = GS_PSMZ_16;  break;
   588  			case GS_PSM_CT16S: gsGlobal->PSMZ = GS_PSMZ_16S; break;
   589  			default:
   590  				printf("gsKit_hires: ERROR: invalid PSM\n");
   591  				return;
   592  		}
   593  	}
   594  
   595  #ifdef BG_DEBUG
   596  	if (gsGlobal->Width < 1024)
   597  		gsGlobal->Width -=  64;
   598  	else
   599  		gsGlobal->Width -= 128;
   600  #endif
   601  
   602  	// Bytes per pixel and height align for render buffers
   603  	if ((gsGlobal->PSM == GS_PSM_CT32) || (gsGlobal->PSM == GS_PSM_CT24)) {
   604  		iBpp = 4;
   605  		iHeightAlign = 32;
   606  	}
   607  	else {
   608  		iBpp = 2;
   609  		iHeightAlign = 64;
   610  	}
   611  
   612  	/*
   613  	 * GS framebuffer storage uses the rounded hardware pitch.  Source
   614  	 * textures/RFB rows continue to use gsGlobal->Width.
   615  	 */
   616  	iPitchPixels = ps2vnc_hires_pitch_pixels(gsGlobal);
   617  
   618  	// The size of a pass must be a multiple of 32 or 64 lines depending on the PSM
   619  	iPassHeight = (gsGlobal->Height + (__iPassCount-1)) / __iPassCount;
   620  	iPassHeight = (iPassHeight + (iHeightAlign-1)) & ~(iHeightAlign-1);
   621  
   622  	// Double buffering is not used in multi pass rendering
   623  	// Force double buffering off
   624  	gsGlobal->DoubleBuffering = GS_SETTING_OFF;
   625  
   626  	// gsKit_init_screen must only allocate 1 front buffer
   627  	// so temporarily disable the depth buffer
   628  	ZBuffering_backup = gsGlobal->ZBuffering;
   629  	gsGlobal->ZBuffering = GS_SETTING_OFF;
   630  
   631  	// This initialize the display and allocates 1 front buffer
   632  	gsKit_init_screen(gsGlobal);
   633  
   634  	/*
   635  	 * Stock gsKit_init_screen() also truncates DISPFB width with Width / 64.
   636  	 * Reassert both read circuits with the same rounded pitch used by FRAME.
   637  	 */
   638  	GS_SET_DISPFB1(
   639  		0,
   640  		iPitchPixels / 64,
   641  		gsGlobal->PSM,
   642  		0,
   643  		0
   644  	);
   645  
   646  	GS_SET_DISPFB2(
   647  		0,
   648  		iPitchPixels / 64,
   649  		gsGlobal->PSM,
   650  		0,
   651  		0
   652  	);
   653  
   654  	// We split the front buffer in different passes
   655  	// Each pass needs to be aligned, so the total alignment requirements become:
   656  	// - __iPassCount * 32 for 32/24 bit color buffers
   657  	// - __iPassCount * 64 for 16    bit color buffers
   658  	{
   659  		u32 allocated_front_size =
   660  			gsKit_texture_size(
   661  				gsGlobal->Width,
   662  				gsGlobal->Height,
   663  				gsGlobal->PSM
   664  			);
   665  
   666  		u32 required_front_size =
   667  			gsKit_texture_size(
   668  				iPitchPixels,
   669  				iPassHeight * __iPassCount,
   670  				gsGlobal->PSM
   671  			);
   672  
   673  		if (allocated_front_size < required_front_size) {
   674  			int iAlign =
   675  				(int)(required_front_size - allocated_front_size);
   676  
   677  			printf(
   678  				"gsKit_hires: Padding pitched front buffer with %dKiB \n",
   679  				iAlign / 1024
   680  			);
   681  
   682  			gsKit_vram_alloc(
   683  				gsGlobal,
   684  				iAlign,
   685  				GSKIT_ALLOC_SYSBUFFER
   686  			);
   687  		}
   688  	}
   689  
   690  	// Restore depth buffer setting
   691  	gsGlobal->ZBuffering = ZBuffering_backup;
   692  	if (gsGlobal->ZBuffering == GS_SETTING_ON)
   693  		gsKit_set_test(gsGlobal, GS_ZTEST_ON);
   694  
   695  	//
   696  	// The front buffer is split into the number of passes
   697  	// Fill the buffer array here
   698  	//
   699  	iBufferline = 0;
   700  	for (iPass = 0; iPass < __iPassCount; iPass++) {
   701  		// Starting address of buffer
   702  		__buffer[iPass].addr = iBufferline * iPitchPixels * iBpp;
   703  
   704  		// First and last line of buffer
   705  		__buffer[iPass].bufferline_begin = iBufferline;
   706  		iBufferline += iPassHeight;
   707  		if (iBufferline > gsGlobal->Height)
   708  			iBufferline = gsGlobal->Height;
   709  		__buffer[iPass].bufferline_end = iBufferline - 1;
   710  
   711  		// First and last line of buffer, measured in CRTC scanlines
   712  		__buffer[iPass].scanline_begin = __buffer[iPass].bufferline_begin;
   713  		__buffer[iPass].scanline_end   = __buffer[iPass].bufferline_end;
   714  		if (gsGlobal->MagV > 0)
   715  		{
   716  			// Lines in the buffer are magnified (displayed more than once
   717  			__buffer[iPass].scanline_begin *= (gsGlobal->MagV+1);
   718  			__buffer[iPass].scanline_end   *= (gsGlobal->MagV+1);
   719  		}
   720  		if ((gsGlobal->Interlace == GS_INTERLACED) && (gsGlobal->Field == GS_FIELD))
   721  		{
   722  			// Only half of the lines in the buffer are read in interlaced field mode
   723  			__buffer[iPass].scanline_begin /= 2;
   724  			__buffer[iPass].scanline_end   /= 2;
   725  		}
   726  	}
   727  
   728  	if ((gsGlobal->ZBuffering == GS_SETTING_ON) && (__iPassCount == 2)) {
   729  		// Depth buffer not inside front buffer, so allocate it
   730  		// We only need half the height of the depth buffer
   731  		gsGlobal->ZBuffer = gsKit_vram_alloc(gsGlobal, gsKit_texture_size(gsGlobal->Width, gsGlobal->Height/2, gsGlobal->PSMZ), GSKIT_ALLOC_SYSBUFFER);
   732  
   733  		// 3rd buffer is fixed to this depth buffer
   734  		__buffer[2].addr = gsGlobal->ZBuffer;
   735  	}
   736  
   737  	// first useable address for textures (after any potential Zbuffer allocation)
   738  	gsGlobal->TexturePointer = gsGlobal->CurrentPointer;
   739  
   740  	//
   741  	// What buffers need to be used in what pass? That's what the pass array is for.
   742  	// Fill the pass array
   743  	//
   744  	for (iPass = 0; iPass < __iPassCount; iPass++) {
   745  		int iPrevPass = (iPass == 0) ? (__iPassCount-1) : (iPass-1);
   746  
   747  		// We always draw top to bottom
   748  		__pass[iPass].DrawBuffer = &__buffer[iPass];
   749  		// The CRTC is right behind the draw buffer
   750  		__pass[iPass].CRTCBuffer = &__buffer[iPrevPass];
   751  
   752  		if (__iPassCount == 2) {
   753  			// Depth buffer fixed
   754  			__pass[iPass].DepthBuffer = &__buffer[2];
   755  			__pass[iPass].TextBuffer  = NULL;
   756  		}
   757  		else if (__iPassCount == 3) {
   758  			int iNextPass = (iPass == (__iPassCount-1)) ? 0 : (iPass+1);
   759  
   760  			// Depth buffer inside front buffer
   761  			__pass[iPass].DepthBuffer = &__buffer[iNextPass];
   762  			__pass[iPass].TextBuffer  = NULL;
   763  		}
   764  		else {
   765  			int iNextPass1 = (iPass == (__iPassCount-1)) ? 0 : (iPass+1);
   766  			int iNextPass2 = (iNextPass1 == (__iPassCount-1)) ? 0 : (iNextPass1+1);
   767  
   768  			// Depth and texture buffers inside front buffer
   769  			//
   770  			// Toggle the position of the buffers relative to the draw buffer
   771  			// so the texture buffer can stay in place twice as long
   772  			__pass[iPass].DepthBuffer = &__buffer[((iPass & 1) == 0) ? iNextPass1 : iNextPass2];
   773  			__pass[iPass].TextBuffer  = &__buffer[((iPass & 1) == 0) ? iNextPass2 : iNextPass1];
   774  		}
   775  	}
   776  
   777  	// (Re-)create pass queues
   778  	_gsKit_create_passes(gsGlobal);
   779  
   780  	// Re-use the "per" queue from gsKit
   781  	__DrawQueue[0] = *gsGlobal->Per_Queue;
   782  	// Create a second identical queue for double buffering
   783  	gsKit_queue_init(gsGlobal, &__DrawQueue[1], GS_PERSISTENT, GS_RENDER_QUEUE_PER_POOLSIZE);
   784  	// Set active queue
   785  	gsKit_queue_set(gsGlobal, &__DrawQueue[__iCurrentDrawQueue]);
   786  
   787  	// Initialize semaphores
   788  	sema.init_count = 0;
   789  	sema.max_count = 1;
   790  	sema.option = 0;
   791  	if ((__sema_hsync_id = CreateSema(&sema)) < 0)
   792  		printf("gsKit_hires: ERROR: creating __sema_hsync_id\n");
   793  	sema.init_count = 0;
   794  	sema.max_count = 1;
   795  	sema.option = 0;
   796  	if ((__sema_vsync_id = CreateSema(&sema)) < 0)
   797  		printf("gsKit_hires: ERROR: creating __sema_vsync_id\n");
   798  	// Initialize mutex (also a semaphore)
   799  	sema.init_count = 1;
   800  	sema.max_count = 1;
   801  	sema.option = 0;
   802  	if ((__sema_queue_id = CreateSema(&sema)) < 0)
   803  		printf("gsKit_hires: ERROR: creating __sema_queue_id\n");
   804  
   805  	// Create thread for transmitting the draw queues to the GS
   806  	// This thread must have the highest priority (1).
   807  	thread.initial_priority	= 1;
   808  	thread.stack_size		= STACK_SIZE;
   809  	thread.gp_reg			= &_gp;
   810  	thread.func				= (void *)dma_to_gs_thread;
   811  	thread.stack			= (void *)__ThreadStack;
   812  	thread.option			= 0;
   813  	thread.attr				= 0;
   814  	if ((__ThreadId = CreateThread(&thread)) < 0)
   815  		printf("gsKit_hires: ERROR: creating thread\n");
   816  	StartThread(__ThreadId, gsGlobal);
   817  
   818  	// HACK: Change the priority of the caller,
   819  	// so we can have the highest priority.
   820  	ChangeThreadPriority(GetThreadId(), 2);
   821  
   822  	// Install hsync handler
   823  	// This will start firing events (using semaphores) to the GS
   824  	if ((__hsync_callback_id = gsKit_add_hsync_handler(hsync_callback)) < 0)
   825  		printf("gsKit_hires: ERROR: registering hsync handler\n");
   826  }

   100   * The System menu is global over ORDINARY local UI, but it must never

   109      if (display_restore_ack_visible ||

   113          display_calibration_request != DISPLAY_MODE_REQUEST_NONE ||

   121  /*
   122   * Return nonzero only when COOLDOWN actually expires.
   123   */
   124   int ps2vnc_system_refresh_update_state(void)
   125  {
   126      unsigned int elapsed_us;
   127  
   128      if (system_refresh_state != SYSTEM_REFRESH_COOLDOWN)
   129          return 0;
   130  
   131      elapsed_us =
   132          profile_ticks_us(
   133              GetTimerSystemTime() -
   134              system_refresh_cooldown_started_tick);
   135  
   136      if (elapsed_us <
   137          SYSTEM_REFRESH_COOLDOWN_SECONDS * 1000000u)
   138          return 0;
   139  
   140      system_refresh_state = SYSTEM_REFRESH_READY;
   141      system_refresh_cooldown_started_tick = 0;
   142      system_menu_status = SYSTEM_MENU_STATUS_NONE;
   143  
   144      if (system_menu_visible)
   145          osk_generation++;
   146  
   147      return 1;
   148  }

   370   * Wait for libpad to finish an asynchronous mode change.
   371   *
   372   * The PS2SDK pad sample waits for STABLE/FINDCTP1 after
   373   * padSetMainMode().  Give this controller up to ~2 seconds.
   374   */
   375   int wait_pad_ready_after_mode_change(void)
   376  {
   377      const struct timespec delay = { 0, 10000000 }; /* 10 ms */
   378      int i;
   379  
   380      for (i = 0; i < 200; i++) {
   381          int state = padGetState(0, 0);
   382  
   383          if (state == PAD_STATE_STABLE ||
   384              state == PAD_STATE_FINDCTP1)
   385              return 0;
   386  
   387          nanosleep(&delay, NULL);
   388      }
   389  
   390      return -1;
   391  }

   394  /*
   395   * Startup-only display recalibration chord.
   396   *
   397   * The normal libpad button bits are active-low, matching the controller
   398   * thread's existing 0xFFFF ^ buttons.btns decoding.
   399   *
   400   * Normal startup returns immediately when L1+R1 are not held.  If both
   401   * buttons are already down, require about 300 ms of continuous hold before
   402   * accepting the override.
   403   */
   404   int startup_force_calibration_requested(void)
   405  {
   406      const struct timespec delay = { 0, 10000000 }; /* 10 ms */
   407      struct padButtonStatus buttons;
   408      unsigned int pressed;
   409      int consecutive = 0;
   410      int saw_valid_sample = 0;
   411      int i;
   412  
   413      for (i = 0; i < 50; i++) {
   414  
   415          if (padGetState(0, 0) == PAD_STATE_STABLE &&
   416              padRead(0, 0, &buttons) != 0) {
   417  
   418              pressed = 0xFFFFu ^ (unsigned int)buttons.btns;
   419  
   420              if (!saw_valid_sample) {
   421                  saw_valid_sample = 1;
   422  
   423                  if ((pressed & (PAD_L1 | PAD_R1)) !=
   424                      (PAD_L1 | PAD_R1))
   425                      return 0;
   426              }
   427  
   428              if ((pressed & (PAD_L1 | PAD_R1)) ==
   429                  (PAD_L1 | PAD_R1)) {
   430  
   431                  consecutive++;
   432  
   433                  if (consecutive >= 30)
   434                      return 1;
   435  
   436              } else {
   437                  return 0;
   438              }
   439          }
   440  
   441          nanosleep(&delay, NULL);
   442      }
   443  
   444      return 0;
   445  }

   483  int rfb_connect_and_handshake_limited(
   484      unsigned int max_width,
   485      unsigned int max_height)
   486  {
   487      static const char client_version[12] = "RFB 003.008\n";
   488  
   489      int sock = -1;
   490      int i;
   491      int none_found = 0;
   492      struct sockaddr_in server;
   493  
   494      char server_version[13];
   495      unsigned char security_count;
   496      unsigned char security_types[255];
   497      unsigned char security_choice = 1;
   498      unsigned char security_result[4];
   499      unsigned char shared_flag = 1;
   500      unsigned char server_init[24];
   501  
   502      unsigned int width;
   503      unsigned int height;
   504      unsigned int name_len;
   505      unsigned int name_take;
   506      unsigned int remaining;
   507  
   508      char desktop_name[128];
   509      unsigned char discard[64];
   510  
   511      screenf(
   512          "Connecting to %s:%d...",
   513          PS2VNC_BOOTSTRAP_PI_IP,
   514          VNC_PORT
   515      );
   516  
   517      sock = socket(AF_INET, SOCK_STREAM, 0);
   518  
   519      if (sock < 0) {
   520          screenf("ERROR: socket() failed: %d", sock);
   521          goto fail;
   522      }
   523  
   524      memset(&server, 0, sizeof(server));
   525      server.sin_len = sizeof(server);
   526      server.sin_family = AF_INET;
   527      server.sin_port = htons(VNC_PORT);
   528      server.sin_addr.s_addr =
   529          inet_addr(PS2VNC_BOOTSTRAP_PI_IP);
   530  
   531      if (connect(sock, (struct sockaddr *)&server, sizeof(server)) < 0) {
   532          screenf("ERROR: connect() failed");
   533          goto fail;
   534      }
   535  
   536      screenf("TCP connected");
   537  
   538      if (recv_exact(sock, server_version, 12) != 12) {
   539          screenf("ERROR: could not read RFB version");
   540          goto fail;
   541      }
   542  
   543      server_version[12] = '\0';
   544  
   545      if (server_version[11] == '\n')
   546          server_version[11] = '\0';
   547  
   548      sanitize_text(server_version);
   549      screenf("Server version: %s", server_version);
   550  
   551      if (send_exact(sock, client_version, 12) != 12) {
   552          screenf("ERROR: could not send RFB version");
   553          goto fail;
   554      }
   555  
   556      if (recv_exact(sock, &security_count, 1) != 1) {
   557          screenf("ERROR: no security-type count");
   558          goto fail;
   559      }
   560  
   561      if (security_count == 0) {
   562          unsigned char lenbuf[4];
   563          unsigned int reason_len;
   564          unsigned int take;
   565          char reason[128];
   566  
   567          if (recv_exact(sock, lenbuf, 4) != 4) {
   568              screenf("ERROR: server rejected connection");
   569              goto fail;
   570          }
   571  
   572          reason_len = read_be32(lenbuf);
   573          take = reason_len;
   574  
   575          if (take > sizeof(reason) - 1)
   576              take = sizeof(reason) - 1;
   577  
   578          if (take > 0 && recv_exact(sock, reason, take) != (int)take) {
   579              screenf("ERROR: failed reading rejection");
   580              goto fail;
   581          }
   582  
   583          reason[take] = '\0';
   584          sanitize_text(reason);
   585  
   586          remaining = reason_len - take;
   587  
   588          while (remaining > 0) {
   589              unsigned int chunk = remaining;
   590  
   591              if (chunk > sizeof(discard))
   592                  chunk = sizeof(discard);
   593  
   594              if (recv_exact(sock, discard, chunk) != (int)chunk)
   595                  break;
   596  
   597              remaining -= chunk;
   598          }
   599  
   600          screenf("SERVER REJECTED: %s", reason);
   601          goto fail;
   602      }
   603  
   604      if (recv_exact(sock, security_types, security_count) != security_count) {
   605          screenf("ERROR: failed reading security types");
   606          goto fail;
   607      }
   608  
   609      for (i = 0; i < security_count; i++) {
   610          if (security_types[i] == 1) {
   611              none_found = 1;
   612              break;
   613          }
   614      }
   615  
   616      if (!none_found) {
   617          screenf("ERROR: SecurityType None not offered");
   618          goto fail;
   619      }
   620  
   621      if (send_exact(sock, &security_choice, 1) != 1) {
   622          screenf("ERROR: could not select security type");
   623          goto fail;
   624      }
   625  
   626      if (recv_exact(sock, security_result, 4) != 4) {
   627          screenf("ERROR: no SecurityResult");
   628          goto fail;
   629      }
   630  
   631      if (read_be32(security_result) != 0) {
   632          screenf("ERROR: authentication failed");
   633          goto fail;
   634      }
   635  
   636      if (send_exact(sock, &shared_flag, 1) != 1) {
   637          screenf("ERROR: could not send ClientInit");
   638          goto fail;
   639      }
   640  
   641      if (recv_exact(sock, server_init, sizeof(server_init)) != sizeof(server_init)) {
   642          screenf("ERROR: failed reading ServerInit");
   643          goto fail;
   644      }
   645  
   646      width = read_be16(&server_init[0]);
   647      height = read_be16(&server_init[2]);
   648      name_len = read_be32(&server_init[20]);
   649  
   650      screenf("Handshake OK - desktop %ux%u", width, height);
   651  
   652      /*
   653       * H4C3C2A:
   654       * The caller explicitly owns the ServerInit acceptance envelope.
   655       *
   656       * Ordinary callers still use OUTPUT_WIDTH/OUTPUT_HEIGHT through the
   657       * wrapper below. Startup reconciliation will later be allowed to use
   658       * the larger allocated RFB framebuffer capacity long enough to repair
   659       * a stale TigerVNC geometry.
   660       *
   661       * Never permit an envelope larger than PS2VNC's backing framebuffer.
   662       */
   663      if (max_width == 0 ||
   664          max_height == 0 ||
   665          max_width > VNC_WIDTH ||
   666          max_height > VNC_HEIGHT) {
   667  
   668          screenf(
   669              "ERROR: invalid RFB limit %ux%u",
   670              max_width,
   671              max_height
   672          );
   673          goto fail;
   674      }
   675  
   676      if (width == 0 || height == 0 ||
   677          width > max_width ||
   678          height > max_height) {
   679  
   680          screenf(
   681              "ERROR: unsupported VNC desktop %ux%u",
   682              width,
   683              height
   684          );
   685          goto fail;
   686      }
   687  
   688      if (desktop_geometry_locked) {
   689          if (width != desktop_width ||
   690              height != desktop_height) {
   691  
   692              screenf(
   693                  "ERROR: expected %ux%u VNC desktop",
   694                  desktop_width,
   695                  desktop_height
   696              );
   697              goto fail;
   698          }
   699      } else {
   700          /*
   701           * A prior interrupted resize may have left TigerVNC smaller than
   702           * 1280x720. Treat the server's actual geometry as authoritative
   703           * until calibration locks the final safe desktop.
   704           */
   705          desktop_width = width;
   706          desktop_height = height;
   707      }
   708  
   709      name_take = name_len;
   710  
   711      if (name_take > sizeof(desktop_name) - 1)
   712          name_take = sizeof(desktop_name) - 1;
   713  
   714      if (name_take > 0) {
   715          if (recv_exact(sock, desktop_name, name_take) != (int)name_take) {
   716              screenf("ERROR: failed reading desktop name");
   717              goto fail;
   718          }
   719      }
   720  
   721      desktop_name[name_take] = '\0';
   722      sanitize_text(desktop_name);
   723  
   724      remaining = name_len - name_take;
   725  
   726      while (remaining > 0) {
   727          unsigned int chunk = remaining;
   728  
   729          if (chunk > sizeof(discard))
   730              chunk = sizeof(discard);
   731  
   732          if (recv_exact(sock, discard, chunk) != (int)chunk) {
   733              screenf("ERROR: failed discarding long name");
   734              goto fail;
   735          }
   736  
   737          remaining -= chunk;
   738      }
   739  
   740      screenf("Desktop name: %.55s", desktop_name);
   741  
   742      return sock;
   743  
   744  fail:
   745      if (sock >= 0)
   746          close(sock);
   747  
   748      return -1;
   749  }

   758  int rfb_connect_and_handshake(void)

   762          OUTPUT_HEIGHT);

   901   * preserved from the server's ExtendedDesktopSize capability response.

   903  static int send_set_desktop_size_and_probe(

   905      unsigned int width,
   906      unsigned int height)
   907  {
   908      unsigned char msg[34];
   909  
   910      memset(msg, 0, sizeof(msg));
   911  
   912      /*
   913       * Bytes 0..23:
   914       * RFB SetDesktopSize.
   915       */
   916      msg[0] = 251;
   917  
   918      msg[2] = (unsigned char)((width >> 8) & 0xFF);
   919      msg[3] = (unsigned char)(width & 0xFF);
   920  
   921      msg[4] = (unsigned char)((height >> 8) & 0xFF);
   922      msg[5] = (unsigned char)(height & 0xFF);
   923  
   924      msg[6] = 1; /* one screen */
   925  
   926      msg[8]  =
   927          (unsigned char)((rfb_first_screen_id >> 24) & 0xFF);
   928      msg[9]  =
   929          (unsigned char)((rfb_first_screen_id >> 16) & 0xFF);
   930      msg[10] =
   931          (unsigned char)((rfb_first_screen_id >> 8) & 0xFF);
   932      msg[11] =
   933          (unsigned char)(rfb_first_screen_id & 0xFF);
   934  
   935      /* Screen x/y remain 0,0. */
   936  
   937      msg[16] = (unsigned char)((width >> 8) & 0xFF);
   938      msg[17] = (unsigned char)(width & 0xFF);
   939  
   940      msg[18] = (unsigned char)((height >> 8) & 0xFF);
   941      msg[19] = (unsigned char)(height & 0xFF);
   942  
   943      msg[20] =
   944          (unsigned char)((rfb_first_screen_flags >> 24) & 0xFF);
   945      msg[21] =
   946          (unsigned char)((rfb_first_screen_flags >> 16) & 0xFF);
   947      msg[22] =
   948          (unsigned char)((rfb_first_screen_flags >> 8) & 0xFF);
   949      msg[23] =
   950          (unsigned char)(rfb_first_screen_flags & 0xFF);
   951  
   952      /*
   953       * TEST13J-D2 SINGLE-WRITE PROBE:
   954       *
   955       * Bytes 24..33 are a second RFB client message:
   956       * an empty incremental FramebufferUpdateRequest.
   957       *
   958       * TigerVNC therefore receives:
   959       *
   960       *   SetDesktopSize
   961       *   FramebufferUpdateRequest
   962       *
   963       * from one PS2 send sequence.
   964       */
   965      msg[24] = 3; /* FramebufferUpdateRequest */
   966      msg[25] = 1; /* incremental */
   967  
   968      /*
   969       * TEST13J-D2:
   970       * Request one harmless pixel rather than an empty region.
   971       *
   972       * The RFB specification permits an empty request, but this avoids
   973       * depending on the server waking its update machinery for 0x0.
   974       */
   975      msg[26] = 0; /* x = 0 */
   976      msg[27] = 0;
   977      msg[28] = 0; /* y = 0 */
   978      msg[29] = 0;
   979      msg[30] = 0; /* width = 1 */
   980      msg[31] = 1;
   981      msg[32] = 0; /* height = 1 */
   982      msg[33] = 1;
   983  
   984      return send_exact(sock, msg, sizeof(msg)) ==
   985             sizeof(msg) ? 0 : -1;
   986  }

   989  int request_and_confirm_desktop_size(
   990      int sock,
   991      unsigned short *rfb_framebuffer,
   992      unsigned short *gs_framebuffer,
   993      unsigned int width,
   994      unsigned int height)
   995  {
   996      unsigned int response_count;
   997  
   998      /*
   999       * F8J2-B2:
  1000       *
  1001       * SetDesktopSize completion is not necessarily the first EDS
  1002       * notification TigerVNC emits at the new geometry.
  1003       *
  1004       * Hardware packet evidence showed:
  1005       *
  1006       *   reason=0 result=0 requested-size
  1007       *   reason=0 result=0 requested-size
  1008       *   reason=0 result=0 requested-size
  1009       *   reason=1 result=0 requested-size
  1010       *
  1011       * reason=0 is an informational server-side layout notification.
  1012       * reason=1 is the acknowledgement for THIS client's SetDesktopSize.
  1013       *
  1014       * Keep consuming complete FramebufferUpdate messages until the
  1015       * matching client acknowledgement arrives.  Geometry validation
  1016       * remains owned by receive_framebuffer_update() while
  1017       * rfb_resize_expected is set.
  1018       *
  1019       * Bound the number of complete server updates accepted while waiting.
  1020       * recv_exact() retains the existing exit/rollback/recovery escape
  1021       * behavior inside each message.
  1022       */
  1023      const unsigned int response_limit = 16;
  1024  
  1025      rfb_resize_expected = 1;
  1026      rfb_resize_expected_width = width;
  1027      rfb_resize_expected_height = height;
  1028  
  1029      if (send_set_desktop_size_and_probe(
  1030              sock,
  1031              width,
  1032              height) < 0) {
  1033  
  1034          rfb_resize_expected = 0;
  1035  
  1036          /* F8J2-B2 diagnostic: SetDesktopSize/probe send failure. */
  1037          live_error_code = 120;
  1038  
  1039          return -1;
  1040      }
  1041  
  1042      for (response_count = 0;
  1043           response_count < response_limit;
  1044           response_count++) {
  1045  
  1046          /*
  1047           * Each receive consumes one complete FramebufferUpdate.
  1048           * Clear the observation fields so an unrelated update cannot
  1049           * inherit a previous EDS notification.
  1050           */
  1051          rfb_extended_desktop_seen = 0;
  1052          rfb_extended_desktop_reason = 0;
  1053          rfb_extended_desktop_result = 0;
  1054          rfb_extended_desktop_width = 0;
  1055          rfb_extended_desktop_height = 0;
  1056          rfb_extended_desktop_screen_count = 0;
  1057  
  1058          if (receive_framebuffer_update(
  1059                  sock,
  1060                  rfb_framebuffer,
  1061                  gs_framebuffer,
  1062                  0,
  1063                  0) < 0) {
  1064  
  1065              rfb_resize_expected = 0;
  1066              return -1;
  1067          }
  1068  
  1069          /*
  1070           * Ordinary framebuffer traffic may be interleaved while the
  1071           * server completes the resize transaction.  It is not the
  1072           * acknowledgement we are waiting for.
  1073           */
  1074          if (!rfb_extended_desktop_seen)
  1075              continue;
  1076  
  1077          /*
  1078           * receive_framebuffer_update() already rejects an EDS rectangle
  1079           * whose dimensions differ from rfb_resize_expected_width/height.
  1080           *
  1081           * A same-geometry reason other than 1 is informational.  Continue
  1082           * until the acknowledgement specifically associated with this
  1083           * client's SetDesktopSize arrives.
  1084           */
  1085          if (rfb_extended_desktop_reason != 1)
  1086              continue;
  1087  
  1088          rfb_resize_expected = 0;
  1089  
  1090          /*
  1091           * reason=1 identifies our request.  result=0 is success.
  1092           */
  1093          if (rfb_extended_desktop_result != 0) {
  1094              live_error_code = 121;
  1095              return -1;
  1096          }
  1097  
  1098          if (rfb_extended_desktop_width != width ||
  1099              rfb_extended_desktop_height != height) {
  1100  
  1101              /*
  1102               * Normally unreachable because the receiver validates this
  1103               * while rfb_resize_expected is active.  Keep an explicit
  1104               * defensive diagnostic at the ownership boundary.
  1105               */
  1106              live_error_code = 118;
  1107              return -1;
  1108          }
  1109  
  1110          return 0;
  1111      }
  1112  
  1113      rfb_resize_expected = 0;
  1114  
  1115      /*
  1116       * We consumed bounded complete server updates without seeing the
  1117       * acknowledgement for our SetDesktopSize.
  1118       */
  1119      live_error_code = 122;
  1120  
  1121      return -1;
  1122  }

  1124   int send_pointer_event(
  1125      int sock,
  1126      unsigned char button_mask,
  1127      unsigned int x,
  1128      unsigned int y)
  1129  {
  1130      unsigned char msg[6];
  1131  
  1132      msg[0] = 5;
  1133      msg[1] = button_mask;
  1134      msg[2] = (unsigned char)((x >> 8) & 0xFF);
  1135      msg[3] = (unsigned char)(x & 0xFF);
  1136      msg[4] = (unsigned char)((y >> 8) & 0xFF);
  1137      msg[5] = (unsigned char)(y & 0xFF);
  1138  
  1139      return send_rfb_message(sock, msg, sizeof(msg)) == sizeof(msg) ? 0 : -1;
  1140  }

  1143  /*
  1144   * RFB KeyEvent:
  1145   *   byte 0     = message type 4
  1146   *   byte 1     = down flag (1 press, 0 release)
  1147   *   bytes 2-3  = padding
  1148   *   bytes 4-7  = X11 keysym, big-endian
  1149   *
  1150   * This uses the same serialized sender as PointerEvent and
  1151   * FramebufferUpdateRequest, so keyboard traffic cannot interleave with
  1152   * either of them on the TCP byte stream.
  1153   */
  1154   int send_key_event(
  1155      int sock,
  1156      unsigned char down,
  1157      unsigned int keysym)
  1158  {
  1159      unsigned char msg[8];
  1160  
  1161      msg[0] = 4;
  1162      msg[1] = down ? 1 : 0;
  1163      msg[2] = 0;
  1164      msg[3] = 0;
  1165      msg[4] = (unsigned char)((keysym >> 24) & 0xFF);
  1166      msg[5] = (unsigned char)((keysym >> 16) & 0xFF);
  1167      msg[6] = (unsigned char)((keysym >> 8) & 0xFF);
  1168      msg[7] = (unsigned char)(keysym & 0xFF);
  1169  
  1170      return send_rfb_message(sock, msg, sizeof(msg)) == sizeof(msg) ? 0 : -1;
  1171  }

  1174   int send_key_tap(int sock, unsigned int keysym)
  1175  {
  1176      unsigned char msg[16];
  1177  
  1178      /* Key down. */
  1179      msg[0] = 4;
  1180      msg[1] = 1;
  1181      msg[2] = 0;
  1182      msg[3] = 0;
  1183      msg[4] = (unsigned char)((keysym >> 24) & 0xFF);
  1184      msg[5] = (unsigned char)((keysym >> 16) & 0xFF);
  1185      msg[6] = (unsigned char)((keysym >> 8) & 0xFF);
  1186      msg[7] = (unsigned char)(keysym & 0xFF);
  1187  
  1188      /* Key up. */
  1189      msg[8]  = 4;
  1190      msg[9]  = 0;
  1191      msg[10] = 0;
  1192      msg[11] = 0;
  1193      msg[12] = msg[4];
  1194      msg[13] = msg[5];
  1195      msg[14] = msg[6];
  1196      msg[15] = msg[7];
  1197  
  1198      return send_rfb_message(sock, msg, sizeof(msg)) == sizeof(msg) ? 0 : -1;
  1199  }

  1202  /*
  1203   * Send a key with optional real Ctrl/Alt modifiers as one serialized RFB
  1204   * write.  This keeps the entire shortcut atomic relative to framebuffer
  1205   * requests and mouse events.
  1206   *
  1207   * Shift remains a local OSK layer; it changes the final printable keysym.
  1208   */
  1209   int send_modified_key_tap(
  1210      int sock,
  1211      unsigned int keysym,
  1212      int use_ctrl,
  1213      int use_alt)
  1214  {
  1215      unsigned char msg[48];
  1216      int pos = 0;
  1217  
  1218  #define APPEND_KEY_EVENT(DOWN, KEYSYM) do {                       \
  1219          unsigned int _k = (unsigned int)(KEYSYM);                 \
  1220          msg[pos + 0] = 4;                                         \
  1221          msg[pos + 1] = (DOWN) ? 1 : 0;                            \
  1222          msg[pos + 2] = 0;                                         \
  1223          msg[pos + 3] = 0;                                         \
  1224          msg[pos + 4] = (unsigned char)((_k >> 24) & 0xFF);        \
  1225          msg[pos + 5] = (unsigned char)((_k >> 16) & 0xFF);        \
  1226          msg[pos + 6] = (unsigned char)((_k >> 8) & 0xFF);         \
  1227          msg[pos + 7] = (unsigned char)(_k & 0xFF);                \
  1228          pos += 8;                                                  \
  1229      } while (0)
  1230  
  1231      if (use_ctrl)
  1232          APPEND_KEY_EVENT(1, XK_CONTROL_L);
  1233  
  1234      if (use_alt)
  1235          APPEND_KEY_EVENT(1, XK_ALT_L);
  1236  
  1237      APPEND_KEY_EVENT(1, keysym);
  1238      APPEND_KEY_EVENT(0, keysym);
  1239  
  1240      if (use_alt)
  1241          APPEND_KEY_EVENT(0, XK_ALT_L);
  1242  
  1243      if (use_ctrl)
  1244          APPEND_KEY_EVENT(0, XK_CONTROL_L);
  1245  
  1246  #undef APPEND_KEY_EVENT
  1247  
  1248      return send_rfb_message(sock, msg, pos) == pos ? 0 : -1;
  1249  }

  1472      int x,
  1473      int y,
  1474      int w,
  1475      int h,
  1476      unsigned short color)
  1477  {

  1480  
  1481      if (x < 0) {
  1482          w += x;
  1483          x = 0;

  1485  
  1486      if (y < 0) {
  1487          h += y;
  1488          y = 0;

  1490  
  1491      if (x + w > OSK_WIDTH)
  1492          w = OSK_WIDTH - x;

  1495          h = OSK_HEIGHT - y;
  1496

  1501          unsigned short *row =
  1502              &osk_pixels[yy * OSK_WIDTH + x];
  1503  
  1504          for (xx = 0; xx < w; xx++)

  1601   void osk_render_bitmap(void)
  1602  {
  1603      const unsigned short panel = osk_color(3, 3, 4);
  1604      const unsigned short key = osk_color(10, 10, 11);
  1605      const unsigned short selected = osk_color(31, 24, 3);
  1606      const unsigned short active = osk_color(6, 20, 8);
  1607      const unsigned short border = osk_color(22, 22, 22);
  1608      const unsigned short text_col = osk_color(31, 31, 31);
  1609      const unsigned short selected_text = osk_color(1, 1, 1);
  1610  
  1611      int row;
  1612      int col;
  1613      int top = 4;
  1614      int row_h = 33;
  1615  
  1616      osk_fill_rect(0, 0, OSK_WIDTH, OSK_HEIGHT, panel);
  1617  
  1618      for (row = 0; row < OSK_ROWS; row++) {
  1619          int count = osk_row_length(row);
  1620          int margin = 6;
  1621          int gap = 3;
  1622          int available =
  1623              OSK_WIDTH - 2 * margin - (count - 1) * gap;
  1624          int cell_w = available / count;
  1625          int used_w =
  1626              cell_w * count + gap * (count - 1);
  1627          int row_x = (OSK_WIDTH - used_w) / 2;
  1628          int y = top + row * row_h;
  1629          int cell_h = row_h - 4;
  1630  
  1631          for (col = 0; col < count; col++) {
  1632              int x = row_x + col * (cell_w + gap);
  1633  
  1634              int is_selected =
  1635                  (row == osk_row && col == osk_col);
  1636  
  1637              unsigned short bg =
  1638                  is_selected ? selected : key;
  1639  
  1640              unsigned short fg =
  1641                  is_selected ? selected_text : text_col;
  1642  
  1643              char label[8];
  1644              int text_scale;
  1645  
  1646              if (row == 4) {
  1647                  strncpy(
  1648                      label,
  1649                      osk_special_labels[col],
  1650                      sizeof(label) - 1
  1651                  );
  1652  
  1653                  label[sizeof(label) - 1] = '\0';
  1654  
  1655                  /*
  1656                   * Persistent state indication:
  1657                   *
  1658                   * ABC / FUNC = active page
  1659                   * Shift/Ctrl/Alt = latched modifier
  1660                   *
  1661                   * Cursor selection still takes visual priority.
  1662                   */
  1663                  if (!is_selected &&
  1664                      ((col == 0 &&
  1665                        osk_page == OSK_PAGE_ABC) ||
  1666                       (col == 1 &&
  1667                        osk_page == OSK_PAGE_FUNC) ||
  1668                       (col == 2 && osk_shift) ||
  1669                       (col == 3 && osk_ctrl) ||
  1670                       (col == 4 && osk_alt))) {
  1671                      bg = active;
  1672                  }
  1673  
  1674              } else if (osk_page == OSK_PAGE_ABC) {
  1675                  label[0] =
  1676                      osk_display_char(row, col);
  1677                  label[1] = '\0';
  1678  
  1679              } else {
  1680                  strncpy(
  1681                      label,
  1682                      osk_func_labels[row][col],
  1683                      sizeof(label) - 1
  1684                  );
  1685  
  1686                  label[sizeof(label) - 1] = '\0';
  1687              }
  1688  
  1689              /*
  1690               * Long labels use the compact font size.
  1691               * F10/F11/F12 and the short utility keys still fit at 2x.
  1692               */
  1693              text_scale =
  1694                  (strlen(label) >= 5) ? 1 : 2;
  1695  
  1696              osk_fill_rect(
  1697                  x - 1, y - 1,
  1698                  cell_w + 2, cell_h + 2,
  1699                  border
  1700              );
  1701  
  1702              osk_fill_rect(
  1703                  x, y,
  1704                  cell_w, cell_h,
  1705                  bg
  1706              );
  1707  
  1708              osk_draw_text_centered(
  1709                  x, y,
  1710                  cell_w, cell_h,
  1711                  label,
  1712                  text_scale,
  1713                  fg
  1714              );
  1715          }
  1716      }
  1717  }

    35  /*
    36   * D17AL-F8G11B:
    37   *
    38   * gsHires CT16S FRAME/DISPFB presentation width is a 64-pixel-unit
    39   * hardware contract.  The mode registry records intended drawing geometry;
    40   * the renderer derives the compatible width without changing physical
    41   * raster timing or Standard-backend semantics.
    42   *
    43   * Keep horizontal width compatibility separate from vertical full-frame
    44   * geometry.  This distinction is essential for interlaced GS_FRAME modes:
    45   * a 720x288 PAL gsHires drawing surface represents a 720x576 full frame.
    46   * Narrowing that surface horizontally must not collapse its EE backing from
    47   * 576 field-separated raster rows to 288 rows.
    48   */
    49  unsigned int ps2vnc_hires_effective_width(void)
    50  {
    51      unsigned int width;
    52  
    53      if (active_video_mode == NULL)
    54          return 0;
    55  
    56      width = active_video_mode->hires_width;
    57  
    58      if (width == 0)
    59          return 0;
    60  
    61      /*
    62       * No current HIRES mode is narrower than one GS FBW unit. Preserve a
    63       * defensive nonzero result if a future diagnostic ever introduces one.
    64       */
    65      if (width < 64u)
    66          return width;
    67  
    68      return (width / 64u) * 64u;

   586   void rfb_dirty_mark_rect(
   587      unsigned int x,
   588      unsigned int y,
   589      unsigned int w,
   590      unsigned int h)
   591  {
   592      unsigned int row;
   593  
   594      if (w == 0 || h == 0)
   595          return;
   596  
   597      if (!rfb_dirty_valid) {
   598          rfb_dirty_x0 = x;
   599          rfb_dirty_y0 = y;
   600          rfb_dirty_x1 = x + w;
   601          rfb_dirty_y1 = y + h;
   602          rfb_dirty_valid = 1;
   603      } else {
   604          if (x < rfb_dirty_x0)
   605              rfb_dirty_x0 = x;
   606          if (y < rfb_dirty_y0)
   607              rfb_dirty_y0 = y;
   608          if (x + w > rfb_dirty_x1)
   609              rfb_dirty_x1 = x + w;
   610          if (y + h > rfb_dirty_y1)
   611              rfb_dirty_y1 = y + h;
   612      }
   613  
   614      for (row = y; row < y + h; row++) {
   615          if (!rfb_dirty_row_valid[row]) {
   616              rfb_dirty_row_x0[row] = x;
   617              rfb_dirty_row_x1[row] = x + w;
   618              rfb_dirty_row_valid[row] = 1;
   619          } else {
   620              if (x < rfb_dirty_row_x0[row])
   621                  rfb_dirty_row_x0[row] = x;
   622              if (x + w > rfb_dirty_row_x1[row])
   623                  rfb_dirty_row_x1[row] = x + w;
   624          }
   625      }
   626  }

  1271  /*
  1272   * Queue a controller-generated RFB message.
  1273   *
  1274   * Before the controller thread exists, fall back to a direct send.  After
  1275   * the queue is active this function NEVER touches the VNC socket: it only
  1276   * copies the complete RFB message into the producer/consumer ring.
  1277   */
  1278   int send_rfb_message(int sock, const void *data, int len)
  1279  {
  1280      const unsigned char *src = (const unsigned char *)data;
  1281      const struct timespec retry_delay = { 0, 1000000 }; /* 1 ms */
  1282  
  1283      if (rfb_queue_sema < 0)
  1284          return send_exact(sock, data, len);
  1285  
  1286      if (len <= 0 || len > RFB_OUT_MESSAGE_MAX)
  1287          return -1;
  1288  
  1289      /*
  1290       * TEST12B:
  1291       * Never allow mouse/key events generated during an outage to pile
  1292       * up and fire into the replacement VNC session.
  1293       */
  1294      if (!rfb_connected)
  1295          return len;
  1296  
  1297      for (;;) {
  1298          unsigned int slot;
  1299  
  1300          if (WaitSema(rfb_queue_sema) < 0)
  1301              return -1;
  1302  
  1303          if (rfb_out_count < RFB_OUT_QUEUE_CAPACITY) {
  1304              slot = rfb_out_tail;
  1305  
  1306              rfb_out_queue[slot].len = (unsigned int)len;
  1307              memcpy(rfb_out_queue[slot].data, src, len);
  1308  
  1309              rfb_out_tail =
  1310                  (rfb_out_tail + 1) % RFB_OUT_QUEUE_CAPACITY;
  1311              rfb_out_count++;
  1312  
  1313              if (SignalSema(rfb_queue_sema) < 0)
  1314                  return -1;
  1315  
  1316              return len;
  1317          }
  1318  
  1319          if (SignalSema(rfb_queue_sema) < 0)
  1320              return -1;
  1321  
  1322          nanosleep(&retry_delay, NULL);
  1323      }
  1324  }

  1446  /*
  1447   * D17AL-F4:
  1448   *
  1449   * The confirmation UI must remain responsive while an incremental
  1450   * FramebufferUpdateRequest is legitimately waiting for remote damage.
  1451   *
  1452   * RFB_RECEIVE_LOCAL_REDRAW is permitted only at a server-message
  1453   * boundary, before recv_exact() has consumed any byte of that message.
  1454   * It is NOT a socket/protocol error and must never interrupt a partially
  1455   * consumed RFB structure.
  1456   */
  1457  #define RFB_RECEIVE_LOCAL_REDRAW (-2)
  1458  
  1459   volatile int rfb_message_boundary_local_yield = 0;
  1460   volatile int rfb_confirm_live_yield_enabled = 0;
  1461   volatile unsigned int rfb_confirm_wait_generation = 0;
  1462  
  1463   int recv_exact(int sock, void *data, int len)
  1464  {
  1465      unsigned char *dst = (unsigned char *)data;
  1466      int done = 0;
  1467      const struct timespec retry_delay = { 0, 1000000 }; /* 1 ms */
  1468  
  1469      while (done < len) {
  1470          unsigned int available;
  1471          unsigned int need;
  1472          unsigned int take;
  1473  
  1474          if (exit_requested)
  1475              return -1;
  1476  
  1477          if (display_confirm_pending &&
  1478  
  1479              display_confirm_action == DISPLAY_CONFIRM_ACTION_ROLLBACK)
  1480  
  1481              return -1;
  1482  
  1483  
  1484          if (runtime_action_request == RUNTIME_ACTION_REFRESH_RFB) {
  1485              debug_emit_now();
  1486              return -1;
  1487          }
  1488  
  1489          /*
  1490           * D17AL-F4:
  1491           * A benign local redraw has lower priority than exit, rollback,
  1492           * and explicit RFB recovery. Yield only before this recv_exact()
  1493           * has consumed anything and only at a complete server-message
  1494           * boundary.
  1495           */
  1496          if (rfb_message_boundary_local_yield &&
  1497              rfb_confirm_live_yield_enabled &&
  1498              done == 0 &&
  1499              osk_generation != rfb_confirm_wait_generation) {
  1500  
  1501              return RFB_RECEIVE_LOCAL_REDRAW;
  1502          }
  1503  
  1504          /*
  1505           * First satisfy the request from bytes already buffered in EE RAM.
  1506           */
  1507          available = rfb_rx_end - rfb_rx_pos;
  1508  
  1509          if (available > 0) {
  1510              need = (unsigned int)(len - done);
  1511              take = (available < need) ? available : need;
  1512  
  1513              memcpy(
  1514                  dst + done,
  1515                  rfb_rx_buffer + rfb_rx_pos,
  1516                  take
  1517              );
  1518  
  1519              rfb_rx_pos += take;
  1520              done += (int)take;
  1521  
  1522              if (rfb_rx_pos == rfb_rx_end) {
  1523                  rfb_rx_pos = 0;
  1524                  rfb_rx_end = 0;
  1525              }
  1526  
  1527              continue;
  1528          }
  1529  
  1530          /*
  1531           * Buffer is empty.  Refill it with one comparatively large recv().
  1532           *
  1533           * Before the controller queue exists (handshake / initial setup),
  1534           * use an ordinary blocking recv.  Afterwards use nonblocking recv so
  1535           * the same main thread can continue draining controller events while
  1536           * waiting for server data.
  1537           */
  1538          {
  1539              u64 refill_start_tick = GetTimerSystemTime();
  1540  
  1541              for (;;) {
  1542                  int n;
  1543  
  1544              if (exit_requested)
  1545                  return -1;
  1546  
  1547              if (display_confirm_pending &&
  1548  
  1549                  display_confirm_action == DISPLAY_CONFIRM_ACTION_ROLLBACK)
  1550  
  1551                  return -1;
  1552  
  1553  
  1554              if (runtime_action_request == RUNTIME_ACTION_REFRESH_RFB) {
  1555                  debug_emit_now();
  1556                  return -1;
  1557              }
  1558  
  1559              /*
  1560               * The refill loop can remain in EAGAIN indefinitely. Re-check
  1561               * the benign message-boundary condition after all hard actions
  1562               * on every idle poll.
  1563               */
  1564              if (rfb_message_boundary_local_yield &&
  1565                  rfb_confirm_live_yield_enabled &&
  1566                  done == 0 &&
  1567                  osk_generation != rfb_confirm_wait_generation) {
  1568  
  1569                  return RFB_RECEIVE_LOCAL_REDRAW;
  1570              }
  1571  
  1572              /*
  1573               * TEST15B2G:
  1574               * Do not prefill the 32 KiB receive buffer on every socket poll.
  1575               * recv() supplies the authoritative byte count and rfb_rx_end
  1576               * bounds all subsequent reads.  The old Test31 sentinel fill was
  1577               * diagnostic instrumentation in a performance-critical path.
  1578               */
  1579              if (rfb_queue_sema >= 0) {
  1580                  if (flush_rfb_outgoing(sock) < 0)
  1581                      return -1;
  1582  
  1583                  n = recv(
  1584                      sock,
  1585                      rfb_rx_buffer,
  1586                      sizeof(rfb_rx_buffer),
  1587                      MSG_DONTWAIT
  1588                  );
  1589              } else {
  1590                  n = recv(
  1591                      sock,
  1592                      rfb_rx_buffer,
  1593                      sizeof(rfb_rx_buffer),
  1594                      0
  1595                  );
  1596              }
  1597  
  1598              rfb_rx_syscalls++;
  1599              rfb_last_recv_n = n;
  1600              rfb_last_recv_call = rfb_rx_syscalls;
  1601  
  1602              if (n > 0) {
  1603                  u64 elapsed = GetTimerSystemTime() - refill_start_tick;
  1604                  profile_rx_wait_ticks += elapsed;
  1605                  if (profile_hextile_active)
  1606                      profile_hextile_rx_wait_ticks += elapsed;
  1607  
  1608                  rfb_rx_pos = 0;
  1609                  rfb_rx_end = (unsigned int)n;
  1610                  break;
  1611              }
  1612  
  1613              if (n == 0)
  1614                  return -1;
  1615  
  1616              if (rfb_queue_sema < 0)
  1617                  return -1;
  1618  
  1619              if (errno != EAGAIN && errno != EWOULDBLOCK)
  1620                  return -1;
  1621  
  1622              /*
  1623               * TEST12B:
  1624               * A physically lost Ethernet link may leave this TCP socket
  1625               * returning EAGAIN indefinitely when there is no outgoing
  1626               * controller traffic. Detect carrier loss proactively so an
  1627               * idle/frozen desktop enters the normal reconnect supervisor.
  1628               */
  1629              if (!ethGetNetIFLinkStatus())
  1630                  return -1;
  1631  
  1632              if (ps2vnc_remote_display_lease_expired())
  1633                  return -1;
  1634  
  1635              if (rfb_message_boundary_local_yield &&
  1636                  done == 0 &&
  1637                  ps2vnc_remote_control_poll_idle() > 0) {
  1638  
  1639                  return RFB_RECEIVE_LOCAL_REDRAW;
  1640              }
  1641  
  1642              debug_emit_idle_poll();
  1643              nanosleep(&retry_delay, NULL);
  1644              }
  1645          }
  1646      }
  1647  
  1648      return done;
  1649  }
