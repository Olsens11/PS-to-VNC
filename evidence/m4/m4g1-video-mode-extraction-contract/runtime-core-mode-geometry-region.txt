     1	ps2vnc_backend_t ps2vnc_video_mode_default_backend(
     2	    const ps2vnc_video_mode_t *mode)
     3	{
     4	    if (mode == NULL)
     5	        return PS2VNC_BACKEND_NONE;
     6	
     7	    if (!ps2vnc_video_mode_supports_backend(

    35	/*
    36	 * D17AL-F8G11B:
    37	 *
    38	 * gsHires CT16S FRAME/DISPFB presentation width is a 64-pixel-unit
    39	 * hardware contract.  The mode registry records intended drawing geometry;
    40	 * the renderer derives the compatible width without changing physical
    41	 * raster timing or Standard-backend semantics.
    42	 *
    43	 * Keep horizontal width compatibility separate from vertical full-frame
    44	 * geometry.  This distinction is essential for interlaced GS_FRAME modes:
    45	 * a 720x288 PAL gsHires drawing surface represents a 720x576 full frame.
    46	 * Narrowing that surface horizontally must not collapse its EE backing from
    47	 * 576 field-separated raster rows to 288 rows.
    48	 */
    49	unsigned int ps2vnc_hires_effective_width(void)
    50	{
    51	    unsigned int width;
    52	
    53	    if (active_video_mode == NULL)
    54	        return 0;
    55	
    56	    width = active_video_mode->hires_width;
    57	
    58	    if (width == 0)
    59	        return 0;
    60	
    61	    /*
    62	     * No current HIRES mode is narrower than one GS FBW unit. Preserve a
    63	     * defensive nonzero result if a future diagnostic ever introduces one.
    64	     */
    65	    if (width < 64u)
    66	        return width;
    67	
    68	    return (width / 64u) * 64u;
    69	}

    71	
    72	static u64 ps2vnc_hires_full_frame_height(void)
    73	{
    74	    u64 height;
    75	
    76	    if (active_video_mode == NULL ||
    77	        active_video_mode->hires_height == 0) {
    78	
    79	        return 0;
    80	    }
    81	
    82	    height =
    83	        (u64)active_video_mode->hires_height;
    84	
    85	    if (active_video_mode->interlace == GS_INTERLACED &&
    86	        active_video_mode->field == GS_FRAME) {
    87	
    88	        height *= 2u;
    89	    }
    90	
    91	    return height;
    92	}

    94	
    95	static int ps2vnc_hires_requires_horizontal_mapping(void)
    96	{
    97	    unsigned int effective_width;
    98	
    99	    if (active_video_mode == NULL ||
   100	        active_video_mode->raster_width == 0) {
   101	
   102	        return 0;
   103	    }
   104	
   105	    effective_width =
   106	        ps2vnc_hires_effective_width();
   107	
   108	    if (effective_width == 0)
   109	        return 0;
   110	
   111	    return
   112	        effective_width !=
   113	            active_video_mode->raster_width;
   114	}
   115	
   116	
   117	 int ps2vnc_hires_requires_vertical_mapping(void)
   118	{
   119	    u64 full_frame_height;
   120	
   121	    if (active_video_mode == NULL ||
   122	        active_video_mode->raster_height == 0) {
   123	
   124	        return 0;
   125	    }
   126	
   127	    full_frame_height =
   128	        ps2vnc_hires_full_frame_height();
   129	
   130	    if (full_frame_height == 0)
   131	        return 0;
   132	
   133	    return
   134	        full_frame_height !=
   135	            (u64)active_video_mode->raster_height;
   136	}

   138	
   139	int ps2vnc_hires_requires_raster_mapping(void)
   140	{
   141	    return
   142	        ps2vnc_hires_requires_horizontal_mapping() ||
   143	        ps2vnc_hires_requires_vertical_mapping();
   144	}

   148	 * EE presentation backing follows the effective horizontal renderer width.
   149	 * Vertical storage remains a separate property:
   150	 *
   151	 *   - true vertical presentation scaling uses the explicit HIRES height;
   152	 *   - otherwise retain full physical-raster height, including the existing
   153	 *     field-separated representation of interlaced GS_FRAME modes.
   154	 */
   155	unsigned int ps2vnc_hires_storage_width(void)
   156	{
   157	    return ps2vnc_hires_effective_width();
   158	}

   161	unsigned int ps2vnc_hires_storage_height(void)
   162	{
   163	    if (active_video_mode == NULL)
   164	        return 0;
   165	
   166	    if (ps2vnc_hires_requires_vertical_mapping())
   167	        return active_video_mode->hires_height;
   168	
   169	    return active_video_mode->raster_height;
   170	}

   186	/*
   187	 * Convert physical output-raster coordinates into the active renderer's
   188	 * drawing space.
   189	 *
   190	 * Physical raster, renderer framebuffer, and RFB desktop geometry are
   191	 * separate coordinate spaces. Each backend supplies its own draw dimensions.
   192	 */
   193	 int ps2vnc_display_logical_desktop_size(
   194	    unsigned int output_width,
   195	    unsigned int output_height,
   196	    unsigned int *width,
   197	    unsigned int *height)
   198	{
   199	    if (active_video_mode == NULL || width == NULL || height == NULL)
   200	        return -1;
   201	
   202	    if (active_display_backend == PS2VNC_BACKEND_HIRES) {
   203	        *width = output_width;
   204	        *height = output_height;
   205	    } else if (active_display_backend == PS2VNC_BACKEND_STANDARD) {
   206	        *width = active_video_mode->standard_logical_width;
   207	        *height = active_video_mode->standard_logical_height;
   208	    } else {
   209	        return -1;
   210	    }
   211	
   212	    if (*width == 0 || *height == 0 ||
   213	        *width > VNC_WIDTH || *height > VNC_HEIGHT)
   214	        return -1;
   215	
   216	    return 0;
   217	}

   219	float ps2vnc_display_output_x(float x)
   220	{
   221	    unsigned int draw_width;
   222	
   223	    if (active_display_backend == PS2VNC_BACKEND_STANDARD)
   224	        draw_width = active_video_mode->standard_fb_width;
   225	    else if (active_display_backend == PS2VNC_BACKEND_HIRES)
   226	        draw_width = ps2vnc_hires_effective_width();
   227	    else
   228	        return x;
   229	
   230	    if (draw_width == 0 || OUTPUT_WIDTH == 0)
   231	        return x;
   232	
   233	    return x * (float)draw_width / (float)OUTPUT_WIDTH;
   234	}
   235	
   236	float ps2vnc_display_output_y(float y)
   237	{
   238	    unsigned int draw_height;
   239	
   240	    if (active_display_backend == PS2VNC_BACKEND_STANDARD)
   241	        draw_height = active_video_mode->standard_fb_height;
   242	    else if (active_display_backend == PS2VNC_BACKEND_HIRES)
   243	        draw_height = active_video_mode->hires_height;
   244	    else
   245	        return y;
   246	
   247	    if (draw_height == 0 || OUTPUT_HEIGHT == 0)
   248	        return y;
   249	
   250	    return y * (float)draw_height / (float)OUTPUT_HEIGHT;
   251	}
