    15	/*
    16	 * D17AL large display-mode matrix.
    17	 *
    18	 * Raster geometry, renderer drawing geometry, and RFB logical geometry are
    19	 * deliberately independent concepts.
    20	 *
    21	 * TV/DTV
    22	 * -------
    23	 * 480i Standard reconstructs the hardware-proven Test13C contract:
    24	 *     physical timing : NTSC 480i
    25	 *     GS drawing      : 704x232 FRAME
    26	 *     RFB desktop     : 704x464
    27	 *
    28	 * 576i begins as HIRES so its full-frame logical geometry can be exercised
    29	 * without inventing an unproven Standard PAL safe-area contract.
    30	 *
    31	 * VGA
    32	 * ---
    33	 * The initial VGA matrix is HIRES-only. It intentionally covers both widths
    34	 * divisible by the GS 64-pixel FBW unit and widths that are not, giving the
    35	 * mode survey useful presentation-path controls.
    36	 */

   269	int ps2vnc_video_mode_supports_backend(
   270	    const ps2vnc_video_mode_t *mode,
   271	    ps2vnc_backend_t backend)
   272	{
   273	    if (mode == NULL || backend == PS2VNC_BACKEND_NONE)
   274	        return 0;
   275	
   276	    return (mode->allowed_backends & (unsigned int)backend) != 0;
   277	}
