   295   unsigned int display_confirm_remaining_seconds(void)
   296  {
   297      unsigned int elapsed_us;
   298  
   299      if (!display_confirm_pending)
   300          return 0;
   301  
   302      elapsed_us = profile_ticks_us(
   303          GetTimerSystemTime() - display_confirm_started_tick);
   304  
   305      if (elapsed_us >=
   306          DISPLAY_CONFIRM_TIMEOUT_SECONDS * 1000000u)
   307          return 0;
   308  
   309      return DISPLAY_CONFIRM_TIMEOUT_SECONDS -
   310          (elapsed_us / 1000000u);
   311  }
