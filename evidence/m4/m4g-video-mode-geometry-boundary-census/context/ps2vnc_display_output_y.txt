   236  float ps2vnc_display_output_y(float y)
   237  {
   238      unsigned int draw_height;
   239  
   240      if (active_display_backend == PS2VNC_BACKEND_STANDARD)
   241          draw_height = active_video_mode->standard_fb_height;
   242      else if (active_display_backend == PS2VNC_BACKEND_HIRES)
   243          draw_height = active_video_mode->hires_height;
   244      else
   245          return y;
   246  
   247      if (draw_height == 0 || OUTPUT_HEIGHT == 0)
   248          return y;
   249  
   250      return y * (float)draw_height / (float)OUTPUT_HEIGHT;
   251  }
